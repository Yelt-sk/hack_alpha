import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error
from flaml import AutoML

DATA_PATH = "data.csv"
TARGETS = ["target_1", "target_2"]

df = pd.read_csv(DATA_PATH).dropna(subset=TARGETS)
X, Y = df.drop(columns=TARGETS), df[TARGETS]
X_train, X_valid, Y_train, Y_valid = train_test_split(
    X, Y, test_size=0.2, random_state=42
)

for target in TARGETS:
    automl = AutoML()
    automl.fit(
        X_train=X_train, y_train=Y_train[target],
        task="regression", metric="rmse",
        time_budget=600, seed=42, n_jobs=4,
    )
    pred = automl.predict(X_valid)
    print(f"{target} RMSE:",
          mean_squared_error(Y_valid[target], pred) ** 0.5)
