import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error
from lightautoml.automl.presets.tabular_presets import TabularAutoML
from lightautoml.tasks import Task

DATA_PATH = "data.csv"
TARGET = "target"
TIMEOUT = 600
N_THREADS = 4
N_FOLDS = 5
SEED = 42

df = pd.read_csv(DATA_PATH).dropna(subset=[TARGET])
train, valid = train_test_split(df, test_size=0.2, random_state=SEED)

automl = TabularAutoML(
    task=Task("reg", loss="mse"),
    timeout=TIMEOUT,
    cpu_limit=N_THREADS,
    reader_params={
        "n_jobs": N_THREADS,
        "cv": N_FOLDS,
        "random_state": SEED,
    },
)
automl.fit_predict(train, roles={"target": TARGET}, verbose=1)
pred = automl.predict(valid).data.ravel()
print("RMSE:", mean_squared_error(valid[TARGET], pred) ** 0.5)
