import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error
from lightautoml.automl.presets.tabular_presets import TabularAutoML
from lightautoml.tasks import Task
from lightautoml.addons.multitarget import MultiautogradSequence

DATA_PATH = "data.csv"
TARGETS = ["target_1", "target_2"]
SEED = 42

df = pd.read_csv(DATA_PATH).dropna(subset=TARGETS)
train, valid = train_test_split(df, test_size=0.2, random_state=SEED)

base = TabularAutoML(
    task=Task("reg", loss="mse"),
    timeout=600,
    cpu_limit=4,
    reader_params={"n_jobs": 4, "cv": 5, "random_state": SEED},
)
automl = MultiautogradSequence(base)
automl.fit_predict(train, roles={"target": TARGETS}, verbose=1)

pred = np.asarray(automl.predict(valid).data)
if pred.ndim == 1:
    pred = pred.reshape(len(valid), len(TARGETS))

for i, target in enumerate(TARGETS):
    rmse = mean_squared_error(valid[target], pred[:, i]) ** 0.5
    print(f"{target} RMSE: {rmse:.6f}")
