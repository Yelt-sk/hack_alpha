import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import f1_score, accuracy_score
from autogluon.tabular import TabularPredictor

DATA_PATH = "data.csv"
TARGET = "target"

df = pd.read_csv(DATA_PATH).dropna(subset=[TARGET])
train, valid = train_test_split(
    df, test_size=0.2, random_state=42, stratify=df[TARGET]
)

predictor = TabularPredictor(
    label=TARGET,
    problem_type="multiclass",
    eval_metric="accuracy",
).fit(train_data=train, time_limit=600)

pred = predictor.predict(valid.drop(columns=[TARGET]))
print("Macro-F1:", f1_score(valid[TARGET], pred, average="macro"))
print("Accuracy:", accuracy_score(valid[TARGET], pred))
